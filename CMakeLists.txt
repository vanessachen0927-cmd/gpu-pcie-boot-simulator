cmake_minimum_required(VERSION 3.16)
project(gpu_pcie_boot_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pcie_core src/pcie.cpp)
target_include_directories(pcie_core PUBLIC src)
target_compile_options(pcie_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gpu_boot src/main.cpp)
target_link_libraries(gpu_boot PRIVATE pcie_core)
add_executable(pcie_tests tests/test_pcie.cpp)
target_link_libraries(pcie_tests PRIVATE pcie_core)
enable_testing()
add_test(NAME pcie_tests COMMAND pcie_tests)

add_executable(config_space_tests tests/test_config_space.cpp)
target_link_libraries(config_space_tests PRIVATE pcie_core)
add_test(NAME config_space_tests COMMAND config_space_tests)

add_executable(enumeration_tests tests/test_enumeration.cpp)
target_link_libraries(enumeration_tests PRIVATE pcie_core)
add_test(NAME enumeration_tests COMMAND enumeration_tests)

add_executable(bar_model_tests tests/test_bar_model.cpp)
target_link_libraries(bar_model_tests PRIVATE pcie_core)
add_test(NAME bar_model_tests COMMAND bar_model_tests)

add_executable(capability_tests tests/test_capabilities.cpp)
target_link_libraries(capability_tests PRIVATE pcie_core)
add_test(NAME capability_tests COMMAND capability_tests)
